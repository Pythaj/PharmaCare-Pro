    80→    80→// Stock level progress bar
    81→    81→function StockBar({ current, reorderLevel }: { current: number; reorderLevel: number }) {
    82→    82→  if (current === 0) return null;
    83→    83→  const maxDisplay = Math.max(reorderLevel * 3, 20);
    84→    84→  const pct = Math.min((current / maxDisplay) * 100, 100);
    85→    85→  const isLow = current <= reorderLevel;
    86→    86→  const isCritical = current <= Math.ceil(reorderLevel * 0.3);
    87→    87→
    88→    88→  return (
    89→    89→    <div className="w-full h-1.5 bg-slate-100 rounded-full overflow-hidden mt-2">
    90→    90→      <motion.div
    91→    91→        className="h-full rounded-full"
    92→    92→        initial={{ width: 0 }}
    93→    93→        animate={{ width: `${pct}%` }}
    94→    94→        transition={{ duration: 0.6, ease: 'easeOut' }}
    95→    95→        style={{
    96→    96→          backgroundColor: isCritical ? '#dc2626' : isLow ? '#f59e0b' : '#10b981',
    97→    97→        }}
    98→    98→      />
    99→    99→    </div>
   100→   100→  );
   101→   101→}

   103→   103→// Stock badge with status
   104→   104→function StockBadge({ stock, reorderLevel }: { stock: number; reorderLevel: number }) {
   105→   105→  if (stock === 0) {
   106→   106→    return (
   107→   107→      <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-red-50 text-red-600 text-xs font-semibold">
   108→   108→        <PackageX className="h-3 w-3" />
   109→   109→        <span>Out of Stock</span>
   110→   110→      </div>
   111→   111→    );
   112→   112→  }
   113→   113→  if (stock <= Math.ceil(reorderLevel * 0.3)) {
   114→   114→    return (
   115→   115→      <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-red-50 text-red-600 text-xs font-semibold">
   116→   116→        <AlertTriangle className="h-3 w-3" />
   117→   117→        <span>Only {stock} left!</span>
   118→   118→      </div>
   119→   119→    );
   120→   120→  }
   121→   121→  if (stock <= reorderLevel) {
   122→   122→    return (
   123→   123→      <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-amber-50 text-amber-700 text-xs font-semibold">
   124→   124→        <TrendingDown className="h-3 w-3" />
   125→   125→        <span>{stock} left</span>
   126→   126→      </div>
   127→   127→    );
   128→   128→  }
   129→   129→  return (
   130→   130→    <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-emerald-50 text-emerald-700 text-xs font-semibold">
   131→   131→      <Package className="h-3 w-3" />
   132→   132→      <span>{stock} in stock</span>
   133→   133→    </div>
   134→   134→  );
   135→   135→}
