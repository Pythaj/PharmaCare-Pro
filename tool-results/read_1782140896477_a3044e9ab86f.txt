     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useRef, Fragment } from 'react';
     4→import { Plus, Search, ChevronDown, ChevronRight, Trash2, Tag, TrendingUp, DollarSign, Pencil, ArrowRightLeft, CircleAlert } from 'lucide-react';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Badge } from '@/components/ui/badge';
     9→import { Skeleton } from '@/components/ui/skeleton';
    10→import { Label } from '@/components/ui/label';
    11→import { Switch } from '@/components/ui/switch';
    12→import {
    13→  Table,
    14→  TableBody,
    15→  TableCell,
    16→  TableHead,
    17→  TableHeader,
    18→  TableRow,
    19→} from '@/components/ui/table';
    20→import {
    21→  Dialog,
    22→  DialogContent,
    23→  DialogDescription,
    24→  DialogHeader,
    25→  DialogTitle,
    26→  DialogFooter,
    27→} from '@/components/ui/dialog';
    28→import {
    29→  Select,
    30→  SelectContent,
    31→  SelectItem,
    32→  SelectTrigger,
    33→  SelectValue,
    34→} from '@/components/ui/select';
    35→import {
    36→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    37→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    38→} from '@/components/ui/alert-dialog';
    39→import { toast } from 'sonner';
    40→import { usePermissions } from '@/hooks/use-permissions';
    41→import type { Product, Batch, Category } from '@/types';
    42→
    43→function formatGHS(value: number): string {
    44→  return new Intl.NumberFormat('en-GH', { style: 'currency', currency: 'GHS' }).format(value);
    45→}
    46→
    47→function calcMargin(cost: number, selling: number): number {
    48→  if (!selling || selling <= 0) return 0;
    49→  return ((selling - cost) / selling) * 100;
    50→}
    51→
    52→interface ProductWithStock extends Product {
    53→  totalStock: number;
    54→  batches: (Batch & { currentQty: number })[];
    55→  earliestExpiry?: string | null;
    56→  daysToExpiry?: number | null;
    57→  hasExpiringBatches?: boolean;
    58→  hasExpiredBatches?: boolean;
    59→  stockStatus?: 'in_stock' | 'low_stock' | 'out_of_stock';
    60→  expiryStatus?: 'good' | 'expiring_soon' | 'expired';
    61→}
    62→
    63→export default function ProductsView() {
    64→  const { canManageProducts } = usePermissions();
    65→  const [products, setProducts] = useState<ProductWithStock[]>([]);
    66→  const [categories, setCategories] = useState<Category[]>([]);
    67→  const [search, setSearch] = useState('');
    68→  const [categoryFilter, setCategoryFilter] = useState('all');
    69→  const [loading, setLoading] = useState(true);
    70→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
    71→  const [showAddDialog, setShowAddDialog] = useState(false);
    72→  const [addForm, setAddForm] = useState({
    73→    name: '',
    74→    genericName: '',
    75→    categoryId: '',
    76→    unit: 'pcs',
    77→    reorderLevel: 10,
    78→    defaultCostPrice: 0,
    79→    defaultSellingPrice: 0,
    80→  });
    81→  const [submitting, setSubmitting] = useState(false);
    82→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
    83→  const [productToDelete, setProductToDelete] = useState<ProductWithStock | null>(null);
    84→  const [deleting, setDeleting] = useState(false);
    85→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    86→
    87→  // Edit Price dialog state
    88→  const [showEditPriceDialog, setShowEditPriceDialog] = useState(false);
    89→  const [editPriceProduct, setEditPriceProduct] = useState<ProductWithStock | null>(null);
    90→  const [editCostPrice, setEditCostPrice] = useState(0);
    91→  const [editSellingPrice, setEditSellingPrice] = useState(0);
    92→  const [editApplyToBatches, setEditApplyToBatches] = useState(false);
    93→  const [updatingPrice, setUpdatingPrice] = useState(false);
    94→
    95→  const fetchProducts = useCallback(async (query: string, cat: string) => {
    96→    try {
    97→      let url = `/api/products?search=${encodeURIComponent(query)}`;
    98→      if (cat && cat !== 'all') url += `&categoryId=${cat}`;
    99→      const res = await fetch(url);
   100→      if (res.ok) {
   101→        const data = await res.json();
   102→        setProducts(data.products ?? []);
   103→      }
   104→    } catch { /* silent */ }
   105→  }, []);
   106→
   107→  useEffect(() => {
   108→    async function init() {
   109→      try {
   110→        const [prodRes, catRes] = await Promise.all([
   111→          fetch('/api/products'),
   112→          fetch('/api/categories'),
   113→        ]);
   114→        if (prodRes.ok) {
   115→          const data = await prodRes.json();
   116→          setProducts(data.products ?? []);
   117→        }
   118→        if (catRes.ok) {
   119→          const data = await catRes.json();
   120→          setCategories(data.categories ?? []);
   121→        }
   122→      } catch { /* silent */ }
   123→      setLoading(false);
   124→    }
   125→    init();
   126→  }, []);
   127→
   128→  const handleSearch = (value: string) => {
   129→    setSearch(value);
   130→    if (debounceRef.current) clearTimeout(debounceRef.current);
   131→    debounceRef.current = setTimeout(() => {
   132→      fetchProducts(value, categoryFilter);
   133→    }, 300);
   134→  };
   135→
   136→  const handleCategoryChange = (value: string) => {
   137→    setCategoryFilter(value);
   138→    fetchProducts(search, value);
   139→  };
   140→
   141→  const getProductStatusBadges = (product: ProductWithStock) => {
   142→    const badges: { label: string; className: string }[] = [];
   143→
   144→    if (product.totalStock === 0) {
   145→      badges.push({ label: 'Out of Stock', className: 'bg-red-500 text-white hover:bg-red-500' });
   146→    } else if (product.totalStock <= product.reorderLevel) {
   147→      badges.push({ label: 'Low Stock', className: 'bg-amber-100 text-amber-700 hover:bg-amber-100 border-amber-300' });
   148→    }
   149→
   150→    if (product.hasExpiredBatches) {
   151→      badges.push({ label: 'Expired Batch', className: 'bg-red-100 text-red-700 hover:bg-red-100' });
   152→    } else if (product.hasExpiringBatches) {
   153→      badges.push({ label: 'Expiring Soon', className: 'bg-orange-100 text-orange-700 hover:bg-orange-100 border-orange-300' });
   154→    }
   155→
   156→    if (badges.length === 0) {
   157→      badges.push({ label: 'In Stock', className: 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-emerald-300' });
   158→    }
   159→
   160→    return badges;
   161→  };
   162→
   163→  const isAlertProduct = (product: ProductWithStock) => {
   164→    return product.stockStatus !== 'in_stock' || product.hasExpiredBatches || product.hasExpiringBatches;
   165→  };
   166→
   167→  const handleAddProduct = async () => {
   168→    if (!addForm.name.trim()) { toast.error('Product name is required'); return; }
   169→    setSubmitting(true);
   170→    try {
   171→      const res = await fetch('/api/products', {
   172→        method: 'POST',
   173→        headers: { 'Content-Type': 'application/json' },
   174→        body: JSON.stringify(addForm),
   175→      });
   176→      if (!res.ok) {
   177→        const data = await res.json().catch(() => ({}));
   178→        throw new Error(data.error || 'Failed to add product');
   179→      }
   180→      toast.success('Product added successfully');
   181→      setShowAddDialog(false);
   182→      setAddForm({
   183→        name: '',
   184→        genericName: '',
   185→        categoryId: '',
   186→        unit: 'pcs',
   187→        reorderLevel: 10,
   188→        defaultCostPrice: 0,
   189→        defaultSellingPrice: 0,
   190→      });
   191→      fetchProducts(search, categoryFilter);
   192→    } catch (err) {
   193→      toast.error(err instanceof Error ? err.message : 'Failed to add product');
   194→    } finally {
   195→      setSubmitting(false);
   196→    }
   197→  };
   198→
   199→  const [batches, setBatches] = useState<(Batch & { currentQty: number })[]>([]);
   200→  const [loadingBatches, setLoadingBatches] = useState(false);
   201→
   202→  const handleExpandRow = async (productId: string) => {
   203→    if (expandedRow === productId) {
   204→      setExpandedRow(null);
   205→      return;
   206→    }
   207→    setExpandedRow(productId);
   208→    setLoadingBatches(true);
   209→    try {
   210→      const res = await fetch(`/api/products/${productId}`);
   211→      if (res.ok) {
   212→        const data = await res.json();
   213→        setBatches(data.batches ?? []);
   214→      }
   215→    } catch { /* silent */ }
   216→    setLoadingBatches(false);
   217→  };
   218→
   219→  const handleDeleteProduct = async () => {
   220→    if (!productToDelete) return;
   221→    setDeleting(true);
   222→    try {
   223→      const res = await fetch(`/api/products/${productToDelete.id}`, { method: 'DELETE' });
   224→      if (!res.ok) {
   225→        const data = await res.json().catch(() => ({}));
   226→        throw new Error(data.error || 'Failed to remove product');
   227→      }
   228→      toast.success(`Product "${productToDelete.name}" removed successfully`);
   229→      setShowDeleteDialog(false);
   230→      setProductToDelete(null);
   231→      fetchProducts(search, categoryFilter);
   232→    } catch (err) {
   233→      toast.error(err instanceof Error ? err.message : 'Failed to remove product');
   234→    } finally {
   235→      setDeleting(false);
   236→    }
   237→  };
   238→
   239→  // ---- Edit Price handlers ----
   240→  const openEditPriceDialog = (product: ProductWithStock) => {
   241→    setEditPriceProduct(product);
   242→    setEditCostPrice(product.defaultCostPrice ?? 0);
   243→    setEditSellingPrice(product.defaultSellingPrice ?? 0);
   244→    setEditApplyToBatches(false);
   245→    setShowEditPriceDialog(true);
   246→  };
   247→
   248→  const handleUpdatePrice = async () => {
   249→    if (!editPriceProduct) return;
   250→    setUpdatingPrice(true);
   251→    try {
   252→      const res = await fetch(`/api/products/${editPriceProduct.id}/update-prices`, {
   253→        method: 'PUT',
   254→        headers: { 'Content-Type': 'application/json' },
   255→        body: JSON.stringify({
   256→          defaultCostPrice: editCostPrice,
   257→          defaultSellingPrice: editSellingPrice,
   258→          applyToBatches: editApplyToBatches,
   259→        }),
   260→      });
   261→      if (!res.ok) {
   262→        const data = await res.json().catch(() => ({}));
   263→        throw new Error(data.error || 'Failed to update prices');
   264→      }
   265→      const data = await res.json();
   266→      toast.success(data.message || `Prices updated for "${editPriceProduct.name}"`);
   267→      setShowEditPriceDialog(false);
   268→      setEditPriceProduct(null);
   269→      fetchProducts(search, categoryFilter);
   270→    } catch (err) {
   271→      toast.error(err instanceof Error ? err.message : 'Failed to update prices');
   272→    } finally {
   273→      setUpdatingPrice(false);
   274→    }
   275→  };
   276→
   277→  // ---- Margin display helper ----
   278→  const MarginDisplay = ({ cost, selling, size = 'sm' }: { cost: number; selling: number; size?: 'sm' | 'xs' }) => {
   279→    const margin = calcMargin(cost, selling);
   280→    if (cost === 0 && selling === 0) {
   281→      return <span className={`text-slate-400 ${size === 'xs' ? 'text-xs' : 'text-sm'}`}>—</span>;
   282→    }
   283→    if (selling === 0) {
   284→      return <span className={`text-slate-400 ${size === 'xs' ? 'text-xs' : 'text-sm'}`}>—</span>;
   285→    }
   286→    if (margin > 0) {
   287→      return (
   288→        <span className={`text-emerald-600 font-medium ${size === 'xs' ? 'text-xs' : 'text-sm'}`}>
   289→          <TrendingUp className="inline h-3 w-3 mr-0.5 -mt-0.5" />
   290→          {margin.toFixed(1)}% <span className="font-normal opacity-75">Profit</span>
   291→        </span>
   292→      );
   293→    }
   294→    if (margin < 0) {
   295→      return (
   296→        <span className={`text-red-600 font-medium ${size === 'xs' ? 'text-xs' : 'text-sm'}`}>
   297→          {margin.toFixed(1)}% <span className="font-normal opacity-75">Loss</span>
   298→        </span>
   299→      );
   300→    }
   301→    return <span className={`text-slate-400 ${size === 'xs' ? 'text-xs' : 'text-sm'}`}>0% —</span>;
   302→  };
   303→
   304→  // Number of columns in the main table
   305→  const totalCols = canManageProducts ? 13 : 12;
   306→
   307→  return (
   308→    <div className="space-y-4 p-6">
   309→
   310→      {/* Pricing Insight Bar */}
   311→      {!loading && products.length > 0 && (
   312→        <div className="flex items-center gap-3 flex-wrap">
   313→          {(() => {
   314→            const priced = products.filter(p => p.defaultSellingPrice > 0).length;
   315→            const unpriced = products.length - priced;
   316→            const avgMargin = products.filter(p => p.defaultSellingPrice > 0 && p.defaultCostPrice > 0)
   317→              .reduce((sum, p) => sum + calcMargin(p.defaultCostPrice, p.defaultSellingPrice), 0) / Math.max(priced, 1);
   318→            return (
   319→              <>
   320→                <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-emerald-50 border border-emerald-100">
   321→                  <DollarSign className="h-3.5 w-3.5 text-emerald-500" />
   322→                  <span className="text-xs font-medium text-emerald-700">{priced} priced</span>
   323→                </div>
   324→                {unpriced > 0 && (
   325→                  <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-amber-50 border border-amber-100">
   326→                    <CircleAlert className="h-3.5 w-3.5 text-amber-500" />
   327→                    <span className="text-xs font-medium text-amber-700">{unpriced} need price</span>
   328→                  </div>
   329→                )}
   330→                {priced > 0 && (
   331→                  <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-slate-50 border border-slate-100">
   332→                    <TrendingUp className="h-3.5 w-3.5 text-slate-400" />
   333→                    <span className="text-xs font-medium text-slate-600">Avg margin: {avgMargin.toFixed(1)}%</span>
   334→                  </div>
   335→                )}
   336→              </>
   337→            );
   338→          })()}
   339→        </div>
   340→      )}
   341→
   342→      <div className="flex flex-col sm:flex-row gap-4 items-start sm:items-center justify-between">
   343→        <div className="relative flex-1 max-w-sm">
   344→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   345→          <Input placeholder="Search products..." value={search} onChange={(e) => handleSearch(e.target.value)} className="pl-10" />
   346→        </div>
   347→        <div className="flex gap-3 items-center">
   348→          <Select value={categoryFilter} onValueChange={handleCategoryChange}>
   349→            <SelectTrigger className="w-44">
   350→              <SelectValue placeholder="All Categories" />
   351→            </SelectTrigger>
   352→            <SelectContent>
   353→              <SelectItem value="all">All Categories</SelectItem>
   354→              {categories.map((cat) => (
   355→                <SelectItem key={cat.id} value={cat.id}>{cat.name}</SelectItem>
   356→              ))}
   357→            </SelectContent>
   358→          </Select>
   359→          {canManageProducts && (
   360→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white shadow-sm shadow-emerald-200/50" onClick={() => setShowAddDialog(true)}>
   361→              <Plus className="h-4 w-4 mr-1.5" />
   362→              Add Product
   363→            </Button>
   364→          )}
   365→        </div>
   366→      </div>
   367→
   368→      <Card>
   369→        <CardContent className="p-0">
   370→          <div className="max-h-[500px] overflow-y-auto">
   371→            <Table>
   372→              <TableHeader>
   373→                <TableRow>
   374→                  <TableHead className="w-8"></TableHead>
   375→                  <TableHead>Name</TableHead>
   376→                  <TableHead>Category</TableHead>
   377→                  <TableHead className="text-right">Stock</TableHead>
   378→                  <TableHead className="text-right">Cost Price</TableHead>
   379→                  <TableHead className="text-right">
   380→                    <div className="flex items-center justify-end gap-1">
   381→                      Selling Price
   382→                      <DollarSign className="h-3 w-3 text-emerald-500" />
   383→                    </div>
   384→                  </TableHead>
   385→                  <TableHead className="text-center">Margin</TableHead>
   386→                  <TableHead>Expiry</TableHead>
   387→                  <TableHead>Status</TableHead>
   388→                  {canManageProducts && <TableHead className="w-20 text-center">Actions</TableHead>}
   389→                </TableRow>
   390→              </TableHeader>
   391→              <TableBody>
   392→                {loading ? (
   393→                  Array.from({ length: 8 }).map((_, i) => (
   394→                    <TableRow key={i}>
   395→                      <TableCell><Skeleton className="h-4 w-4" /></TableCell>
   396→                      <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   397→                      <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   398→                      <TableCell><Skeleton className="h-4 w-14 ml-auto" /></TableCell>
   399→                      <TableCell><Skeleton className="h-4 w-18 ml-auto" /></TableCell>
   400→                      <TableCell><Skeleton className="h-4 w-16 ml-auto" /></TableCell>
   401→                      <TableCell><Skeleton className="h-4 w-14 mx-auto" /></TableCell>
   402→                      <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   403→                      <TableCell><Skeleton className="h-5 w-24" /></TableCell>
   404→                      {canManageProducts && <TableCell><Skeleton className="h-4 w-16 mx-auto" /></TableCell>}
   405→                    </TableRow>
   406→                  ))
   407→                ) : products.length > 0 ? (
   408→                  products.map((product, index) => {
   409→                    const badges = getProductStatusBadges(product);
   410→                    const hasAlert = isAlertProduct(product);
   411→                    const isExpanded = expandedRow === product.id;
   412→                    const rowKey = product.id || `product-${index}`;
   413→                    return (
   414→                      <Fragment key={rowKey}>
   415→                        <TableRow
   416→                          className={`cursor-pointer hover:bg-muted/50 ${
   417→                            hasAlert
   418→                              ? product.totalStock === 0
   419→                                ? 'bg-red-50/40'
   420→                                : product.hasExpiredBatches
   421→                                  ? 'bg-red-50/20'
   422→                                  : 'bg-amber-50/20'
   423→                              : ''
   424→                          }`}
   425→                          onClick={() => handleExpandRow(rowKey)}
   426→                        >
   427→                          <TableCell>
   428→                            {isExpanded ? <ChevronDown className="h-4 w-4" /> : <ChevronRight className="h-4 w-4" />}
   429→                          </TableCell>
   430→                          <TableCell>
   431→                            <div className="flex flex-col gap-0.5">
   432→                              <span className="font-medium text-sm leading-tight">{product.name}</span>
   433→                              {product.genericName && (
   434→                                <span className="text-[11px] text-muted-foreground leading-tight">{product.genericName}</span>
   435→                              )}
   436→                              <span className="text-[10px] text-slate-400">{product.unit}</span>
   437→                            </div>
   438→                          </TableCell>
   439→                          <TableCell><Badge variant="outline" className="text-[11px]">{product.category?.name ?? '-'}</Badge></TableCell>
   440→                          <TableCell className="text-right">
   441→                            <span className={`font-mono font-semibold text-sm ${
   442→                              product.totalStock === 0 ? 'text-red-600' :
   443→                              product.totalStock <= product.reorderLevel ? 'text-amber-600' :
   444→                              'text-slate-900'
   445→                            }`}>
   446→                              {product.totalStock}
   447→                            </span>
   448→                          </TableCell>
   449→                          <TableCell className="text-right">
   450→                            {product.defaultCostPrice > 0 ? (
   451→                              <span className="text-sm text-slate-600 font-mono">{formatGHS(product.defaultCostPrice)}</span>
   452→                            ) : (
   453→                              <span className="text-slate-300 text-xs">—</span>
   454→                            )}
   455→                          </TableCell>
   456→                          <TableCell className="text-right">
   457→                            {product.defaultSellingPrice > 0 ? (
   458→                              <button
   459→                                className="group/price flex items-center justify-end gap-1 hover:bg-emerald-50 -my-1 px-1.5 py-1 rounded-md transition-colors"
   460→                                onClick={(e) => { e.stopPropagation(); openEditPriceDialog(product); }}
   461→                                title="Click to edit price"
   462→                              >
   463→                                <span className="font-bold text-emerald-700 text-sm">{formatGHS(product.defaultSellingPrice)}</span>
   464→                                <Pencil className="h-3 w-3 text-emerald-400 opacity-0 group-hover/price:opacity-100 transition-opacity" />
   465→                              </button>
   466→                            ) : (
   467→                              <button
   468→                                className="group/price flex items-center justify-center gap-1 hover:bg-amber-50 -my-1 px-2 py-1 rounded-md transition-colors"
   469→                                onClick={(e) => { e.stopPropagation(); openEditPriceDialog(product); }}
   470→                                title="Click to set price"
   471→                              >
   472→                                <CircleAlert className="h-3.5 w-3.5 text-amber-400" />
   473→                                <span className="text-xs text-amber-500 font-medium">Set Price</span>
   474→                                <Pencil className="h-3 w-3 text-amber-300 opacity-0 group-hover/price:opacity-100 transition-opacity" />
   475→                              </button>
   476→                            )}
   477→                          </TableCell>
   478→                          <TableCell className="text-center">
   479→                            <MarginDisplay cost={product.defaultCostPrice} selling={product.defaultSellingPrice} size="xs" />
   480→                          </TableCell>
   481→                          <TableCell>
   482→                            {product.earliestExpiry ? (
   483→                              <div className="flex items-center gap-1">
   484→                                <div className={`w-1.5 h-1.5 rounded-full ${
   485→                                  product.daysToExpiry !== null && product.daysToExpiry < 0 ? 'bg-red-500' :
   486→                                  product.daysToExpiry !== null && product.daysToExpiry < 30 ? 'bg-red-500' :
   487→                                  product.daysToExpiry !== null && product.daysToExpiry < 90 ? 'bg-amber-500' :
   488→                                  'bg-emerald-500'
   489→                                }`} />
   490→                                <span className="text-xs">{new Date(product.earliestExpiry).toLocaleDateString('en-GH')}</span>
   491→                              </div>
   492→                            ) : (
   493→                              <span className="text-xs text-muted-foreground">-</span>
   494→                            )}
   495→                          </TableCell>
   496→                          <TableCell>
   497→                            <div className="flex flex-wrap gap-1">
   498→                              {badges.map((b) => (
   499→                                <Badge
   500→                                  key={b.label}
   501→                                  variant="outline"
   502→                                  className={`text-[10px] px-1.5 py-0 h-5 ${b.className}`}

   599→                                            </td>

   603→                                    </tbody>
   604→                                  </table>
   605→                                  {canManageProducts && batches.length > 0 && (

   609→                                    </p>
   610→                                  )}
   611→                                </div>
   612→                              ) : (
   613→                                <p className="text-sm text-muted-foreground">No batches found</p>
   614→                              )}
   615→                            </TableCell>
   616→                          </TableRow>
   617→                        )}
   618→                      </Fragment>
   619→                    );
   620→                  })
   621→                ) : (
   622→                  <TableRow>
   623→                    <TableCell colSpan={totalCols} className="text-center text-muted-foreground py-12">
   624→                      No products found
   625→                    </TableCell>
   626→                  </TableRow>
   627→                )}
   628→              </TableBody>
   629→            </Table>
   630→          </div>
   631→        </CardContent>
   632→      </Card>
   633→

   638→            name: '',
   639→            genericName: '',
   640→            categoryId: '',
   641→            unit: 'pcs',
   642→            reorderLevel: 10,
   643→            defaultCostPrice: 0,
   644→            defaultSellingPrice: 0,
   645→          });
   646→        }
   647→        setShowAddDialog(open);
   648→      }}>
   649→        <DialogContent className="sm:max-w-lg">
   650→          <DialogHeader>

   655→              Add New Product

   657→            <DialogDescription>Enter product details and set pricing to make it immediately sellable at the POS.</DialogDescription>

   659→          <div className="space-y-4">
   660→            <div>

   663→            </div>
   664→            <div>
   665→              <Label>Generic Name</Label>

   680→              <div>

   701→            {/* Premium Pricing Section */}
   702→            <div className="relative rounded-xl border-2 border-emerald-200 bg-gradient-to-br from-emerald-50/80 via-white to-teal-50/60 p-5 space-y-4 shadow-sm shadow-emerald-100/50">
   703→              <div className="relative flex items-center gap-2.5">
   704→                <div className="flex items-center justify-center h-8 w-8 rounded-lg bg-gradient-to-br from-emerald-500 to-emerald-600 text-white shadow-sm shadow-emerald-200/50">
   705→                  <DollarSign className="h-4 w-4" />
   706→                </div>

   735→                  />

   809→                      min="0"

   820→                    <MarginDisplay cost={editCostPrice} selling={editSellingPrice} size="sm" />
   821→                  </div>
   822→                </div>
   823→              </div>
   824→
   825→              {/* Apply to batches toggle */}
   826→              {(editPriceProduct._count?.batches ?? 0) > 0 && (

   855→            </Button>
   856→          </DialogFooter>
   857→        </DialogContent>
   858→      </Dialog>
   859→
   860→      {/* Delete Product Confirmation */}
   861→      <AlertDialog open={showDeleteDialog} onOpenChange={setShowDeleteDialog}>
   862→        <AlertDialogContent>

   866→              This will deactivate the product and hide it from all listings. Existing sales records containing
   867→              this product will remain unaffected.
   868→            </AlertDialogDescription>
   869→          </AlertDialogHeader>
   870→          <AlertDialogFooter>
   871→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   872→            <AlertDialogAction
   873→              className="bg-red-600 hover:bg-red-700 text-white"
   874→              onClick={handleDeleteProduct}
   875→              disabled={deleting}
   876→            >
   877→              {deleting ? 'Removing...' : 'Remove Product'}
   878→            </AlertDialogAction>
   879→          </AlertDialogFooter>
   880→        </AlertDialogContent>
   881→      </AlertDialog>
   882→    </div>
   883→  );
