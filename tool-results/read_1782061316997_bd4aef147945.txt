    98→    98→// ─── Helpers: flatten / unflatten settings for API ─────────────
    99→    99→
   100→   100→function flattenSettings(s: AllSettings): Record<string, string> {
   101→   101→  const result: Record<string, string> = {};
   102→   102→  const walk = (obj: Record<string, unknown>, prefix: string) => {
   103→   103→    for (const [k, v] of Object.entries(obj)) {
   104→   104→      const key = prefix ? `${prefix}.${k}` : k;
   105→   105→      if (v !== null && typeof v === 'object' && !Array.isArray(v)) {
   106→   106→        walk(v as Record<string, unknown>, key);
   107→   107→      } else {
   108→   108→        result[key] = String(v);
   109→   109→      }
   110→   110→    }
   111→   111→  };
   112→   112→  walk(s as unknown as Record<string, unknown>, '');
   113→   113→  return result;
   114→   114→}
   115→   115→
   116→   116→function unflattenSettings(flat: Record<string, string>): Partial<AllSettings> {
   117→   117→  const result: Record<string, unknown> = {};
   118→   118→  for (const [compoundKey, value] of Object.entries(flat)) {
   119→   119→    const parts = compoundKey.split('.');
   120→   120→    let current: Record<string, unknown> = result;
   121→   121→    for (let i = 0; i < parts.length - 1; i++) {
   122→   122→      if (!current[parts[i]] || typeof current[parts[i]] !== 'object') {
   123→   123→        current[parts[i]] = {};
   124→   124→      }
   125→   125→      current = current[parts[i]] as Record<string, unknown>;
   126→   126→    }
   127→   127→    // Parse numbers and booleans
   128→   128→    let parsed: string | number | boolean = value;
   129→   129→    if (value === 'true') parsed = true;
   130→   130→    else if (value === 'false') parsed = false;
   131→   131→    else if (value !== '' && !isNaN(Number(value))) parsed = Number(value);
   132→   132→    current[parts[parts.length - 1]] = parsed;
   133→   133→  }
   134→   134→  return result as unknown as Partial<AllSettings>;
   135→   135→}
