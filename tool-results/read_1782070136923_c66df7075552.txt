    99→    99→// ─── Helpers: flatten / unflatten settings for API ─────────────
   100→   100→
   101→   101→function flattenSettings(s: AllSettings): Record<string, string> {
   102→   102→  const result: Record<string, string> = {};
   103→   103→  const walk = (obj: Record<string, unknown>, prefix: string) => {
   104→   104→    for (const [k, v] of Object.entries(obj)) {
   105→   105→      const key = prefix ? `${prefix}.${k}` : k;
   106→   106→      if (v !== null && typeof v === 'object' && !Array.isArray(v)) {
   107→   107→        walk(v as Record<string, unknown>, key);
   108→   108→      } else {
   109→   109→        result[key] = String(v);
   110→   110→      }
   111→   111→    }
   112→   112→  };
   113→   113→  walk(s as unknown as Record<string, unknown>, '');
   114→   114→  return result;
   115→   115→}
   116→   116→
   117→   117→function unflattenSettings(flat: Record<string, string>): Partial<AllSettings> {
   118→   118→  const result: Record<string, unknown> = {};
   119→   119→  for (const [compoundKey, value] of Object.entries(flat)) {
   120→   120→    const parts = compoundKey.split('.');
   121→   121→    let current: Record<string, unknown> = result;
   122→   122→    for (let i = 0; i < parts.length - 1; i++) {
   123→   123→      if (!current[parts[i]] || typeof current[parts[i]] !== 'object') {
   124→   124→        current[parts[i]] = {};
   125→   125→      }
   126→   126→      current = current[parts[i]] as Record<string, unknown>;
   127→   127→    }
   128→   128→    // Parse numbers and booleans
   129→   129→    let parsed: string | number | boolean = value;
   130→   130→    if (value === 'true') parsed = true;
   131→   131→    else if (value === 'false') parsed = false;
   132→   132→    else if (value !== '' && !isNaN(Number(value))) parsed = Number(value);
   133→   133→    current[parts[parts.length - 1]] = parsed;
   134→   134→  }
   135→   135→  return result as unknown as Partial<AllSettings>;
   136→   136→}
