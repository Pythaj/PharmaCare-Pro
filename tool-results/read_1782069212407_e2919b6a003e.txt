     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react';
     4→import {
     5→  Search,
     6→  Plus,
     7→  Minus,
     8→  Trash2,
     9→  ShoppingCart,
    10→  CreditCard,
    11→  Smartphone,
    12→  Banknote,
    13→  X,
    14→  Printer,
    15→  CheckCircle,
    16→  UserRoundPlus,
    17→  Zap,
    18→} from 'lucide-react';
    19→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    20→import { Button } from '@/components/ui/button';
    21→import { Input } from '@/components/ui/input';
    22→import { Badge } from '@/components/ui/badge';
    23→import { Separator } from '@/components/ui/separator';
    24→import { Skeleton } from '@/components/ui/skeleton';
    25→import { ScrollArea } from '@/components/ui/scroll-area';
    26→import {
    27→  Dialog,
    28→  DialogContent,
    29→  DialogHeader,
    30→  DialogTitle,
    31→} from '@/components/ui/dialog';
    32→import { useAppStore } from '@/stores/app-store';
    33→import { toast } from 'sonner';
    34→import type { Product, Batch, Customer, CartItem } from '@/types';
    35→
    36→function formatGHS(value: number): string {
    37→  return new Intl.NumberFormat('en-GH', { style: 'currency', currency: 'GHS' }).format(value);
    38→}
    39→
    40→interface ProductWithStock extends Product {
    41→  batches: (Batch & { currentQty: number })[];
    42→  totalStock: number;
    43→  minSellingPrice: number;
    44→}
    45→
    46→export default function POSView() {
    47→  const { cart, addToCart, removeFromCart, updateCartQuantity, clearCart, currentUser, selectedCustomerId, setSelectedCustomer } = useAppStore();
    48→  const [searchQuery, setSearchQuery] = useState('');
    49→  const [products, setProducts] = useState<ProductWithStock[]>([]);
    50→  const [categories, setCategories] = useState<string[]>([]);
    51→  const [activeCategory, setActiveCategory] = useState<string>('All');
    52→  const [customers, setCustomers] = useState<Customer[]>([]);
    53→  const [customerSearch, setCustomerSearch] = useState('');
    54→  const [discount, setDiscount] = useState(0);
    55→  const [notes, setNotes] = useState('');
    56→  const [paymentMethod, setPaymentMethod] = useState<'cash' | 'card' | 'mobile_money'>('cash');
    57→  const [showReceipt, setShowReceipt] = useState(false);
    58→  const [completedSale, setCompletedSale] = useState<{
    59→    invoiceNo: string;
    60→    customerName?: string;
    61→    items: { productName: string; quantity: number; unitPrice: number; total: number }[];
    62→    subtotal: number;
    63→    tax: number;
    64→    discount: number;
    65→    totalAmount: number;
    66→    paymentMethod: string;
    67→    createdAt: string;
    68→  } | null>(null);
    69→  const [submitting, setSubmitting] = useState(false);
    70→  const [loading, setLoading] = useState(true);
    71→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    72→
    73→  // Walk-in customer state
    74→  const [showWalkInDialog, setShowWalkInDialog] = useState(false);
    75→  const [walkInName, setWalkInName] = useState('');
    76→  const [walkInPhone, setWalkInPhone] = useState('');
    77→  const [addingWalkIn, setAddingWalkIn] = useState(false);
    78→
    79→  const fetchProducts = useCallback(async (query: string) => {
    80→    try {
    81→      const res = await fetch(`/api/products?search=${encodeURIComponent(query)}`);
    82→      if (res.ok) {
    83→        const data = await res.json();
    84→        setProducts(data.products ?? []);
    85→        const cats = [...new Set((data.products ?? []).map((p: Product) => p.category?.name).filter(Boolean))] as string[];
    86→        setCategories(['All', ...cats]);
    87→      }
    88→    } catch { /* silent */ }
    89→  }, []);
    90→
    91→  useEffect(() => {
    92→    async function init() {
    93→      await fetchProducts('');
    94→      try {
    95→        const res = await fetch('/api/customers');
    96→        if (res.ok) {
    97→          const data = await res.json();
    98→          setCustomers(data.customers ?? []);
    99→        }
   100→      } catch { /* silent */ }
   101→      setLoading(false);
   102→    }
   103→    init();
   104→  }, [fetchProducts]);
   105→
   106→  const handleSearch = (value: string) => {
   107→    setSearchQuery(value);
   108→    if (debounceRef.current) clearTimeout(debounceRef.current);
   109→    debounceRef.current = setTimeout(() => {
   110→      fetchProducts(value);
   111→    }, 300);
   112→  };
   113→
   114→  const handleAddToCart = (product: ProductWithStock) => {
   115→    const bestBatch = product.batches
   116→      .filter(b => b.quantity > 0)
   117→      .sort((a, b) => new Date(a.expiryDate).getTime() - new Date(b.expiryDate).getTime())[0];
   118→    if (!bestBatch) {
   119→      toast.error('No available stock for this product');
   120→      return;
   121→    }
   122→    addToCart({
   123→      productId: product.id,
   124→      productName: product.name,
   125→      batchId: bestBatch.id,
   126→      batchNumber: bestBatch.batchNumber,
   127→      quantity: 1,
   128→      unitPrice: bestBatch.sellingPrice,
   129→      costPrice: bestBatch.costPrice,
   130→      availableQty: bestBatch.currentQty,
   131→      expiryDate: bestBatch.expiryDate,
   132→    });
   133→    toast.success(`${product.name} added to cart`);
   134→  };
   135→
   136→  const subtotal = cart.reduce((sum, item) => sum + item.unitPrice * item.quantity, 0);
   137→  const tax = subtotal * 0.125; // 12.5% VAT
   138→  const total = subtotal + tax - discount;
   139→
   140→  const filteredProducts = activeCategory === 'All'
   141→    ? products
   142→    : products.filter(p => p.category?.name === activeCategory);
   143→
   144→  const handleCompleteSale = async () => {
   145→    if (cart.length === 0) {
   146→      toast.error('Cart is empty');
   147→      return;
   148→    }
   149→    setSubmitting(true);
   150→    try {
   151→      const res = await fetch('/api/sales', {
   152→        method: 'POST',
   153→        headers: { 'Content-Type': 'application/json' },
   154→        body: JSON.stringify({
   155→          userId: currentUser?.id,
   156→          customerId: selectedCustomerId,
   157→          items: cart.map(item => ({
   158→            productId: item.productId,
   159→            batchId: item.batchId,
   160→            quantity: item.quantity,
   161→            unitPrice: item.unitPrice,
   162→            costPrice: item.costPrice,
   163→          })),
   164→          discount,
   165→          paymentMethod,
   166→          notes: notes || undefined,
   167→        }),
   168→      });
   169→      if (!res.ok) {
   170→        const data = await res.json().catch(() => ({}));
   171→        throw new Error(data.error || 'Failed to complete sale');
   172→      }
   173→      const data = await res.json();
   174→      setCompletedSale({
   175→        invoiceNo: data.invoiceNo,
   176→        customerName: data.customer?.name,
   177→        items: (data.items ?? []).map((item: any) => ({
   178→          productName: item.product?.name ?? 'Unknown',
   179→          quantity: item.quantity,
   180→          unitPrice: item.unitPrice,
   181→          total: item.total,
   182→        })),
   183→        subtotal: data.subtotal,
   184→        tax: data.tax,
   185→        discount: data.discount,
   186→        totalAmount: data.totalAmount,
   187→        paymentMethod: data.paymentMethod,
   188→        createdAt: data.createdAt,
   189→      });
   190→      setShowReceipt(true);
   191→      toast.success('Sale completed successfully!');
   192→    } catch (err) {
   193→      toast.error(err instanceof Error ? err.message : 'Failed to complete sale');
   194→    } finally {
   195→      setSubmitting(false);
   196→    }
   197→  };
   198→
   199→  const handleCloseReceipt = () => {
   200→    setShowReceipt(false);
   201→    setCompletedSale(null);
   202→    clearCart();
   203→    setDiscount(0);
   204→    setNotes('');
   205→    setPaymentMethod('cash');
   206→  };
   207→
   208→  const paymentIcons = {
   209→    cash: Banknote,
   210→    card: CreditCard,
   211→    mobile_money: Smartphone,
   212→  };
   213→
   214→  const selectedCustomer = customers.find(c => c.id === selectedCustomerId);
   215→
   216→  if (loading) {
   217→    return (
   218→      <div className="flex h-full gap-4 p-4">
   219→        <div className="flex-1 space-y-4">
   220→          <Skeleton className="h-10 w-full" />
   221→          <div className="grid grid-cols-2 lg:grid-cols-3 gap-3">
   222→            {Array.from({ length: 9 }).map((_, i) => (
   223→              <Skeleton key={i} className="h-28 w-full" />
   224→            ))}
   225→          </div>
   226→        </div>
   227→        <div className="w-96 border-l p-4 space-y-4">
   228→          <Skeleton className="h-8 w-40" />
   229→          {Array.from({ length: 4 }).map((_, i) => (
   230→            <Skeleton key={i} className="h-16 w-full" />
   231→          ))}
   232→          <Skeleton className="h-40 w-full" />
   233→        </div>
   234→      </div>
   235→    );
   236→  }
   237→
   238→  return (
   239→    <div className="flex h-[calc(100vh-5rem)] gap-4 p-4">
   240→      {/* Left Panel - Product Search & Listing */}
   241→      <div className="flex-1 flex flex-col min-w-0">
   242→        {/* Search */}
   243→        <div className="relative mb-4">
   244→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   245→          <Input
   246→            placeholder="Search products by name..."
   247→            value={searchQuery}
   248→            onChange={(e) => handleSearch(e.target.value)}
   249→            className="pl-10"
   250→          />
   251→        </div>
   252→
   253→        {/* Category Tabs */}
   254→        <div className="flex gap-2 mb-4 overflow-x-auto pb-2">
   255→          {categories.map((cat) => (
   256→            <Button
   257→              key={cat}
   258→              variant={activeCategory === cat ? 'default' : 'outline'}
   259→              size="sm"
   260→              className={activeCategory === cat ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   261→              onClick={() => setActiveCategory(cat)}
   262→            >
   263→              {cat}
   264→            </Button>
   265→          ))}
   266→        </div>
   267→
   268→        {/* Product Grid */}
   269→        <ScrollArea className="flex-1">
   270→          <div className="grid grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   271→            {filteredProducts.length > 0 ? (
   272→              filteredProducts.map((product) => (
   273→                <button
   274→                  key={product.id}
   275→                  onClick={() => handleAddToCart(product)}
   276→                  className="text-left p-4 rounded-xl border hover:border-emerald-400 hover:shadow-md transition-all bg-card"
   277→                >
   278→                  <div className="flex items-start justify-between gap-2">
   279→                    <h4 className="font-medium text-sm leading-tight line-clamp-2">{product.name}</h4>
   280→                    <Pill className="h-4 w-4 text-emerald-500 shrink-0 mt-0.5" />
   281→                  </div>
   282→                  <p className="text-xs text-muted-foreground mt-1">{product.category?.name ?? 'Uncategorized'}</p>
   283→                  <div className="flex items-center justify-between mt-3">
   284→                    <span className="font-bold text-emerald-600">{formatGHS(product.minSellingPrice)}</span>
   285→                    <Badge variant={product.totalStock > 0 ? 'default' : 'destructive'} className={
   286→                      product.totalStock > 0 ? 'bg-emerald-100 text-emerald-700' : ''
   287→                    }>
   288→                      {product.totalStock}
   289→                    </Badge>
   290→                  </div>
   291→                </button>
   292→              ))
   293→            ) : (
   294→              <div className="col-span-full text-center text-muted-foreground py-12">
   295→                No products found
   296→              </div>
   297→            )}
   298→          </div>
   299→        </ScrollArea>
   300→      </div>
   301→
   302→      {/* Right Panel - Cart & Checkout */}
   303→      <div className="w-96 border-l bg-card flex flex-col shrink-0">
   304→        <CardHeader className="pb-3">
   305→          <CardTitle className="text-lg flex items-center gap-2">
   306→            <ShoppingCart className="h-5 w-5" />
   307→            Cart ({cart.length} items)
   308→          </CardTitle>
   309→        </CardHeader>
   310→
   311→        {/* Cart Items */}
   312→        <ScrollArea className="flex-1 px-4">
   313→          {cart.length > 0 ? (
   314→            <div className="space-y-3">
   315→              {cart.map((item) => (
   316→                <div key={`${item.productId}-${item.batchId}`} className="p-3 rounded-lg border">
   317→                  <div className="flex items-start justify-between">
   318→                    <div className="flex-1 min-w-0">
   319→                      <p className="font-medium text-sm truncate">{item.productName}</p>
   320→                      <p className="text-xs text-muted-foreground">{formatGHS(item.unitPrice)} each</p>
   321→                    </div>
   322→                    <Button
   323→                      variant="ghost"
   324→                      size="icon"
   325→                      className="h-7 w-7 text-red-400 hover:text-red-600"
   326→                      onClick={() => removeFromCart(item.productId, item.batchId)}
   327→                    >
   328→                      <Trash2 className="h-4 w-4" />
   329→                    </Button>
   330→                  </div>
   331→                  <div className="flex items-center justify-between mt-2">
   332→                    <div className="flex items-center gap-2">
   333→                      <Button
   334→                        variant="outline"
   335→                        size="icon"
   336→                        className="h-7 w-7"
   337→                        onClick={() => updateCartQuantity(item.productId, item.batchId, item.quantity - 1)}
   338→                        disabled={item.quantity <= 1}
   339→                      >
   340→                        <Minus className="h-3 w-3" />
   341→                      </Button>
   342→                      <span className="w-8 text-center font-medium text-sm">{item.quantity}</span>
   343→                      <Button
   344→                        variant="outline"
   345→                        size="icon"
   346→                        className="h-7 w-7"
   347→                        onClick={() => updateCartQuantity(item.productId, item.batchId, item.quantity + 1)}
   348→                        disabled={item.quantity >= item.availableQty}
   349→                      >
   350→                        <Plus className="h-3 w-3" />
   351→                      </Button>
   352→                    </div>
   353→                    <span className="font-bold text-sm">{formatGHS(item.unitPrice * item.quantity)}</span>
   354→                  </div>
   355→                </div>
   356→              ))}
   357→            </div>
   358→          ) : (
   359→            <div className="text-center text-muted-foreground py-12">
   360→              <ShoppingCart className="h-12 w-12 mx-auto mb-3 opacity-30" />
   361→              <p>Cart is empty</p>
   362→              <p className="text-xs">Click a product to add it</p>
   363→            </div>
   364→          )}
   365→        </ScrollArea>
   366→
   367→        {/* Cart Summary & Checkout */}
   368→        <div className="border-t p-4 space-y-3">
   369→          {/* Customer Selection */}
   370→          <div className="space-y-2">
   371→            <div className="relative">
   372→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   373→              <Input
   374→                placeholder="Search customer (optional)"
   375→                className="pl-8 h-8 text-xs"
   376→                value={customerSearch}
   377→                onChange={(e) => setCustomerSearch(e.target.value)}
   378→              />
   379→              {customerSearch && !selectedCustomerId && (
   380→                <div className="absolute top-full left-0 right-0 mt-1 bg-background border rounded-lg shadow-lg max-h-40 overflow-y-auto z-50">
   381→                  {customers
   382→                    .filter(c => c.name.toLowerCase().includes(customerSearch.toLowerCase()) || c.phone?.includes(customerSearch))
   383→                    .slice(0, 5)
   384→                    .map(c => (
   385→                      <button
   386→                        key={c.id}
   387→                        className="w-full text-left px-3 py-2 hover:bg-muted text-sm"
   388→                        onClick={() => { setSelectedCustomer(c.id); setCustomerSearch(c.name); }}
   389→                      >
   390→                        {c.name} - {c.phone ?? c.email ?? '-'}
   391→                      </button>
   392→                    ))
   393→                }
   394→              </div>
   395→              )}
   396→              {selectedCustomer && (
   397→                <div className="flex items-center justify-between mt-1">
   398→                  <span className="text-xs text-emerald-600 font-medium">{selectedCustomer.name}</span>
   399→                  <button onClick={() => { setSelectedCustomer(null); setCustomerSearch(''); }} className="text-xs text-red-500 hover:underline">Remove</button>
   400→                </div>
   401→              )}
   402→            </div>
   403→            {!selectedCustomerId && (
   404→              <Button
   405→                variant="outline"
   406→                size="sm"
   407→                className="w-full h-8 text-xs border-dashed border-emerald-300 text-emerald-600 hover:bg-emerald-50 hover:text-emerald-700 hover:border-emerald-400 transition-all"
   408→                onClick={() => { setWalkInName('Walk-In Customer'); setWalkInPhone(''); setShowWalkInDialog(true); }}
   409→              >
   410→                <Zap className="h-3 w-3 mr-1" />
   411→                Quick Walk-In
   412→              </Button>
   413→            )}
   414→          </div>
   415→
   416→          {/* Notes */}
   417→          <Input
   418→            placeholder="Notes (optional)"
   419→            value={notes}
   420→            onChange={(e) => setNotes(e.target.value)}
   421→            className="h-8 text-xs"
   422→          />
   423→
   424→      {/* Walk-In Customer Dialog */}
   425→      <Dialog open={showWalkInDialog} onOpenChange={(open) => { if (!open) setShowWalkInDialog(false); }}>
   426→        <DialogContent className="sm:max-w-sm">
   427→          <DialogHeader>
   428→            <DialogTitle className="flex items-center gap-2">
   429→              <div className="h-8 w-8 rounded-lg bg-emerald-100 flex items-center justify-center">
   430→                <UserRoundPlus className="h-4 w-4 text-emerald-600" />
   431→              </div>
   432→              Quick Walk-In
   433→            </DialogTitle>
   434→          </DialogHeader>
   435→          <div className="space-y-3">
   436→            <div>
   437→              <Label className="text-xs text-muted-foreground">Customer Name</Label>
   438→              <Input
   439→                value={walkInName}
   440→                onChange={(e) => setWalkInName(e.target.value)}
   441→                placeholder="e.g. Walk-In Customer"
   442→                className="mt-1"
   443→                autoFocus
   444→              />
   445→            </div>
   446→            <div>
   447→              <Label className="text-xs text-muted-foreground">Phone <span className="text-muted-foreground/50">(optional)</span></Label>
   448→              <Input
   449→                value={walkInPhone}
   450→                onChange={(e) => setWalkInPhone(e.target.value)}
   451→                placeholder="e.g. +233 24 123 4567"
   452→                className="mt-1"
   453→                onKeyDown={(e) => { if (e.key === 'Enter') handleAddWalkIn(); }}
   454→              />
   455→            </div>
   456→            <p className="text-xs text-muted-foreground flex items-center gap-1.5">
   457→              <Zap className="h-3 w-3" />
   458→              Walk-in customers are quick-registered for this sale. Details are optional.
   459→            </p>
   460→          </div>
   461→          <DialogFooter className="gap-2 sm:gap-0">
   462→            <Button variant="outline" size="sm" onClick={() => setShowWalkInDialog(false)}>Cancel</Button>
   463→            <Button
   464→              size="sm"
   465→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   466→              onClick={handleAddWalkIn}
   467→              disabled={addingWalkIn || !walkInName.trim()}
   468→            >
   469→              <UserRoundPlus className="h-4 w-4 mr-1" />
   470→              {addingWalkIn ? 'Adding...' : 'Add & Select'}
   471→            </Button>
   472→          </DialogFooter>
   473→        </DialogContent>
   474→      </Dialog>
   475→
   476→          <div className="space-y-1.5 text-sm">
   477→            <div className="flex justify-between">
   478→              <span className="text-muted-foreground">Subtotal</span>
   479→              <span>{formatGHS(subtotal)}</span>
   480→            </div>
   481→            <div className="flex justify-between">
   482→              <span className="text-muted-foreground">Tax (12.5%)</span>
   483→              <span>{formatGHS(tax)}</span>
   484→            </div>
   485→            <div className="flex items-center justify-between">
   486→              <span className="text-muted-foreground">Discount</span>
   487→              <Input
   488→                type="number"
   489→                value={discount || ''}
   490→                onChange={(e) => setDiscount(Number(e.target.value) || 0)}
   491→                className="w-24 h-7 text-xs text-right"
   492→                placeholder="0.00"
   493→              />
   494→            </div>
   495→            <Separator />
   496→            <div className="flex justify-between font-bold text-lg">
   497→              <span>Total</span>
   498→              <span className="text-emerald-600">{formatGHS(total)}</span>
   499→            </div>
   500→          </div>
   501→
   502→          {/* Payment Method */}
   503→          <div className="flex gap-2">
   504→            {(['cash', 'card', 'mobile_money'] as const).map((method) => {
   505→              const Icon = paymentIcons[method];
   506→              const label = method === 'mobile_money' ? 'Mobile Money' : method.charAt(0).toUpperCase() + method.slice(1);
   507→              return (
   508→                <Button
   509→                  key={method}
   510→                  variant={paymentMethod === method ? 'default' : 'outline'}
   511→                  size="sm"
   512→                  className={paymentMethod === method ? 'bg-emerald-600 hover:bg-emerald-700 flex-1' : 'flex-1'}
   513→                  onClick={() => setPaymentMethod(method)}
   514→                >
   515→                  <Icon className="h-4 w-4 mr-1" />
   516→                  {label}
   517→                </Button>
   518→              );
   519→            })}
   520→          </div>
   521→
   522→          {/* Action Buttons */}
   523→          <div className="flex gap-2">
   524→            <Button
   525→              variant="outline"
   526→              className="flex-1"
   527→              onClick={clearCart}
   528→              disabled={cart.length === 0}
   529→            >
   530→              <X className="h-4 w-4 mr-1" />
   531→              Clear
   532→            </Button>
   533→            <Button
   534→              className="flex-1 bg-emerald-600 hover:bg-emerald-700 text-white"
   535→              onClick={handleCompleteSale}
   536→              disabled={cart.length === 0 || submitting}
   537→            >
   538→              <CheckCircle className="h-4 w-4 mr-1" />
   539→              {submitting ? 'Processing...' : 'Complete Sale'}
   540→            </Button>
   541→          </div>
   542→        </div>
   543→      </div>
   544→
   545→      {/* Receipt Modal */}
   546→      <Dialog open={showReceipt} onOpenChange={(open) => { if (!open) handleCloseReceipt(); }}>
   547→        <DialogContent className="max-w-md">
   548→          <DialogHeader>
   549→            <DialogTitle>Sale Completed</DialogTitle>
   550→          </DialogHeader>
   551→          {completedSale && (
