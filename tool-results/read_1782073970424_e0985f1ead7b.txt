   133→// ─── Helpers: flatten / unflatten settings for API ─────────────
   134→
   135→function flattenSettings(s: AllSettings): Record<string, string> {
   136→  const result: Record<string, string> = {};
   137→  const walk = (obj: Record<string, unknown>, prefix: string) => {
   138→    for (const [k, v] of Object.entries(obj)) {
   139→      const key = prefix ? `${prefix}.${k}` : k;
   140→      if (v !== null && typeof v === 'object' && !Array.isArray(v)) {
   141→        walk(v as Record<string, unknown>, key);
   142→      } else {
   143→        result[key] = String(v);
   144→      }
   145→    }
   146→  };
   147→  walk(s as unknown as Record<string, unknown>, '');
   148→  return result;
   149→}
   150→
   151→function unflattenSettings(flat: Record<string, string>): Partial<AllSettings> {
   152→  const result: Record<string, unknown> = {};
   153→  for (const [compoundKey, value] of Object.entries(flat)) {
   154→    const parts = compoundKey.split('.');
   155→    let current: Record<string, unknown> = result;
   156→    for (let i = 0; i < parts.length - 1; i++) {
   157→      if (!current[parts[i]] || typeof current[parts[i]] !== 'object') {
   158→        current[parts[i]] = {};
   159→      }
   160→      current = current[parts[i]] as Record<string, unknown>;
   161→    }
   162→    // Parse numbers and booleans
   163→    let parsed: string | number | boolean = value;
   164→    if (value === 'true') parsed = true;
   165→    else if (value === 'false') parsed = false;
   166→    else if (value !== '' && !isNaN(Number(value))) parsed = Number(value);
   167→    current[parts[parts.length - 1]] = parsed;
   168→  }
   169→  return result as unknown as Partial<AllSettings>;
   170→}
