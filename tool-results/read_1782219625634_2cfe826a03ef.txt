   135→// ─── Helpers: flatten / unflatten settings for API ─────────────
   136→
   137→function flattenSettings(s: AllSettings): Record<string, string> {
   138→  const result: Record<string, string> = {};
   139→  const walk = (obj: Record<string, unknown>, prefix: string) => {
   140→    for (const [k, v] of Object.entries(obj)) {
   141→      const key = prefix ? `${prefix}.${k}` : k;
   142→      if (v !== null && typeof v === 'object' && !Array.isArray(v)) {
   143→        walk(v as Record<string, unknown>, key);
   144→      } else {
   145→        result[key] = String(v);
   146→      }
   147→    }
   148→  };
   149→  walk(s as unknown as Record<string, unknown>, '');
   150→  return result;
   151→}
   152→
   153→function unflattenSettings(flat: Record<string, string>): Partial<AllSettings> {
   154→  const result: Record<string, unknown> = {};
   155→  for (const [compoundKey, value] of Object.entries(flat)) {
   156→    const parts = compoundKey.split('.');
   157→    let current: Record<string, unknown> = result;
   158→    for (let i = 0; i < parts.length - 1; i++) {
   159→      if (!current[parts[i]] || typeof current[parts[i]] !== 'object') {
   160→        current[parts[i]] = {};
   161→      }
   162→      current = current[parts[i]] as Record<string, unknown>;
   163→    }
   164→    // Parse numbers and booleans
   165→    let parsed: string | number | boolean = value;
   166→    if (value === 'true') parsed = true;
   167→    else if (value === 'false') parsed = false;
   168→    else if (value !== '' && !isNaN(Number(value))) parsed = Number(value);
   169→    current[parts[parts.length - 1]] = parsed;
   170→  }
   171→  return result as unknown as Partial<AllSettings>;
   172→}
