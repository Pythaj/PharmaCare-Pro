    98→// ─── Helpers: flatten / unflatten settings for API ─────────────
    99→
   100→function flattenSettings(s: AllSettings): Record<string, string> {
   101→  const result: Record<string, string> = {};
   102→  const walk = (obj: Record<string, unknown>, prefix: string) => {
   103→    for (const [k, v] of Object.entries(obj)) {
   104→      const key = prefix ? `${prefix}.${k}` : k;
   105→      if (v !== null && typeof v === 'object' && !Array.isArray(v)) {
   106→        walk(v as Record<string, unknown>, key);
   107→      } else {
   108→        result[key] = String(v);
   109→      }
   110→    }
   111→  };
   112→  walk(s as unknown as Record<string, unknown>, '');
   113→  return result;
   114→}
   115→
   116→function unflattenSettings(flat: Record<string, string>): Partial<AllSettings> {
   117→  const result: Record<string, unknown> = {};
   118→  for (const [compoundKey, value] of Object.entries(flat)) {
   119→    const parts = compoundKey.split('.');
   120→    let current: Record<string, unknown> = result;
   121→    for (let i = 0; i < parts.length - 1; i++) {
   122→      if (!current[parts[i]] || typeof current[parts[i]] !== 'object') {
   123→        current[parts[i]] = {};
   124→      }
   125→      current = current[parts[i]] as Record<string, unknown>;
   126→    }
   127→    // Parse numbers and booleans
   128→    let parsed: string | number | boolean = value;
   129→    if (value === 'true') parsed = true;
   130→    else if (value === 'false') parsed = false;
   131→    else if (value !== '' && !isNaN(Number(value))) parsed = Number(value);
   132→    current[parts[parts.length - 1]] = parsed;
   133→  }
   134→  return result as unknown as Partial<AllSettings>;
   135→}
