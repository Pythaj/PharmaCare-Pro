    65→// Animated stock counter component
    66→function AnimatedStockNumber({ value, changed }: { value: number; changed: boolean }) {
    67→  return (
    68→    <motion.span
    69→      key={value}
    70→      initial={changed ? { scale: 1.3, color: '#dc2626' } : false}
    71→      animate={{ scale: 1, color: 'inherit' }}
    72→      transition={{ duration: 0.5, ease: 'easeOut' }}
    73→      className="font-bold tabular-nums"
    74→    >
    75→      {value}
    76→    </motion.span>
    77→  );
    78→}

    80→// Stock level progress bar
    81→function StockBar({ current, reorderLevel }: { current: number; reorderLevel: number }) {
    82→  if (current === 0) return null;
    83→  const maxDisplay = Math.max(reorderLevel * 3, 20);
    84→  const pct = Math.min((current / maxDisplay) * 100, 100);
    85→  const isLow = current <= reorderLevel;
    86→  const isCritical = current <= Math.ceil(reorderLevel * 0.3);
    87→
    88→  return (
    89→    <div className="w-full h-1.5 bg-slate-100 rounded-full overflow-hidden mt-2">
    90→      <motion.div
    91→        className="h-full rounded-full"
    92→        initial={{ width: 0 }}
    93→        animate={{ width: `${pct}%` }}
    94→        transition={{ duration: 0.6, ease: 'easeOut' }}
    95→        style={{
    96→          backgroundColor: isCritical ? '#dc2626' : isLow ? '#f59e0b' : '#10b981',
    97→        }}
    98→      />
    99→    </div>
   100→  );
   101→}

   103→// Stock badge with status
   104→function StockBadge({ stock, reorderLevel }: { stock: number; reorderLevel: number }) {
   105→  if (stock === 0) {
   106→    return (
   107→      <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-red-50 text-red-600 text-xs font-semibold">
   108→        <PackageX className="h-3 w-3" />
   109→        <span>Out of Stock</span>
   110→      </div>
   111→    );
   112→  }
   113→  if (stock <= Math.ceil(reorderLevel * 0.3)) {
   114→    return (
   115→      <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-red-50 text-red-600 text-xs font-semibold">
   116→        <AlertTriangle className="h-3 w-3" />
   117→        <span>Only {stock} left!</span>
   118→      </div>
   119→    );
   120→  }
   121→  if (stock <= reorderLevel) {
   122→    return (
   123→      <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-amber-50 text-amber-700 text-xs font-semibold">
   124→        <TrendingDown className="h-3 w-3" />
   125→        <span>{stock} left</span>
   126→      </div>
   127→    );
   128→  }
   129→  return (
   130→    <div className="flex items-center gap-1 px-2 py-0.5 rounded-md bg-emerald-50 text-emerald-700 text-xs font-semibold">
   131→      <Package className="h-3 w-3" />
   132→      <span>{stock} in stock</span>
   133→    </div>
   134→  );
   135→}
