     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react';
     4→import {
     5→  Search,
     6→  Plus,
     7→  Minus,
     8→  Trash2,
     9→  ShoppingCart,
    10→  CreditCard,
    11→  Smartphone,
    12→  Banknote,
    13→  X,
    14→  Printer,
    15→  CheckCircle,
    16→  UserRoundPlus,
    17→  Zap,
    18→} from 'lucide-react';
    19→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    20→import { Button } from '@/components/ui/button';
    21→import { Input } from '@/components/ui/input';
    22→import { Badge } from '@/components/ui/badge';
    23→import { Separator } from '@/components/ui/separator';
    24→import { Skeleton } from '@/components/ui/skeleton';
    25→import { ScrollArea } from '@/components/ui/scroll-area';
    26→import {
    27→  Dialog,
    28→  DialogContent,
    29→  DialogFooter,
    30→  DialogHeader,
    31→  DialogTitle,
    32→} from '@/components/ui/dialog';
    33→import { Label } from '@/components/ui/label';
    34→import { useAppStore } from '@/stores/app-store';
    35→import { toast } from 'sonner';
    36→import type { Product, Batch, Customer, CartItem } from '@/types';
    37→
    38→function formatGHS(value: number): string {
    39→  return new Intl.NumberFormat('en-GH', { style: 'currency', currency: 'GHS' }).format(value);
    40→}
    41→
    42→interface ProductWithStock extends Product {
    43→  batches: (Batch & { currentQty: number })[];
    44→  totalStock: number;
    45→  minSellingPrice: number;
    46→}
    47→
    48→export default function POSView() {
    49→  const { cart, addToCart, removeFromCart, updateCartQuantity, clearCart, currentUser, selectedCustomerId, setSelectedCustomer } = useAppStore();
    50→  const [searchQuery, setSearchQuery] = useState('');
    51→  const [products, setProducts] = useState<ProductWithStock[]>([]);
    52→  const [categories, setCategories] = useState<string[]>([]);
    53→  const [activeCategory, setActiveCategory] = useState<string>('All');
    54→  const [customers, setCustomers] = useState<Customer[]>([]);
    55→  const [customerSearch, setCustomerSearch] = useState('');
    56→  const [discount, setDiscount] = useState(0);
    57→  const [notes, setNotes] = useState('');
    58→  const [paymentMethod, setPaymentMethod] = useState<'cash' | 'card' | 'mobile_money'>('cash');
    59→  const [showReceipt, setShowReceipt] = useState(false);
    60→  const [completedSale, setCompletedSale] = useState<{
    61→    invoiceNo: string;
    62→    customerName?: string;
    63→    items: { productName: string; quantity: number; unitPrice: number; total: number }[];
    64→    subtotal: number;
    65→    tax: number;
    66→    discount: number;
    67→    totalAmount: number;
    68→    paymentMethod: string;
    69→    createdAt: string;
    70→  } | null>(null);
    71→  const [submitting, setSubmitting] = useState(false);
    72→  const [loading, setLoading] = useState(true);
    73→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    74→
    75→  // Walk-in customer state
    76→  const [showWalkInDialog, setShowWalkInDialog] = useState(false);
    77→  const [walkInName, setWalkInName] = useState('');
    78→  const [walkInPhone, setWalkInPhone] = useState('');
    79→  const [addingWalkIn, setAddingWalkIn] = useState(false);
    80→
    81→  const fetchProducts = useCallback(async (query: string) => {
    82→    try {
    83→      const res = await fetch(`/api/products?search=${encodeURIComponent(query)}`);
    84→      if (res.ok) {
    85→        const data = await res.json();
    86→        setProducts(data.products ?? []);
    87→        const cats = [...new Set((data.products ?? []).map((p: Product) => p.category?.name).filter(Boolean))] as string[];
    88→        setCategories(['All', ...cats]);
    89→      }
    90→    } catch { /* silent */ }
    91→  }, []);
    92→
    93→  useEffect(() => {
    94→    async function init() {
    95→      await fetchProducts('');
    96→      try {
    97→        const res = await fetch('/api/customers');
    98→        if (res.ok) {
    99→          const data = await res.json();
   100→          setCustomers(data.customers ?? []);
   101→        }
   102→      } catch { /* silent */ }
   103→      setLoading(false);
   104→    }
   105→    init();
   106→  }, [fetchProducts]);
   107→
   108→  const handleSearch = (value: string) => {
   109→    setSearchQuery(value);
   110→    if (debounceRef.current) clearTimeout(debounceRef.current);
   111→    debounceRef.current = setTimeout(() => {
   112→      fetchProducts(value);
   113→    }, 300);
   114→  };
   115→
   116→  const handleAddToCart = (product: ProductWithStock) => {
   117→    const bestBatch = product.batches
   118→      .filter(b => b.quantity > 0)
   119→      .sort((a, b) => new Date(a.expiryDate).getTime() - new Date(b.expiryDate).getTime())[0];
   120→    if (!bestBatch) {
   121→      toast.error('No available stock for this product');
   122→      return;
   123→    }
   124→    addToCart({
   125→      productId: product.id,
   126→      productName: product.name,
   127→      batchId: bestBatch.id,
   128→      batchNumber: bestBatch.batchNumber,
   129→      quantity: 1,
   130→      unitPrice: bestBatch.sellingPrice,
   131→      costPrice: bestBatch.costPrice,
   132→      availableQty: bestBatch.currentQty,
   133→      expiryDate: bestBatch.expiryDate,
   134→    });
   135→    toast.success(`${product.name} added to cart`);
   136→  };
   137→
   138→  const subtotal = cart.reduce((sum, item) => sum + item.unitPrice * item.quantity, 0);
   139→  const tax = subtotal * 0.125; // 12.5% VAT
   140→  const total = subtotal + tax - discount;
   141→
   142→  const filteredProducts = activeCategory === 'All'
   143→    ? products
   144→    : products.filter(p => p.category?.name === activeCategory);
   145→
   146→  const handleCompleteSale = async () => {
   147→    if (cart.length === 0) {
   148→      toast.error('Cart is empty');
   149→      return;
   150→    }
   151→    setSubmitting(true);
   152→    try {
   153→      const res = await fetch('/api/sales', {
   154→        method: 'POST',
   155→        headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({
   157→          userId: currentUser?.id,
   158→          customerId: selectedCustomerId,
   159→          items: cart.map(item => ({
   160→            productId: item.productId,
   161→            batchId: item.batchId,
   162→            quantity: item.quantity,
   163→            unitPrice: item.unitPrice,
   164→            costPrice: item.costPrice,
   165→          })),
   166→          discount,
   167→          paymentMethod,
   168→          notes: notes || undefined,
   169→        }),
   170→      });
   171→      if (!res.ok) {
   172→        const data = await res.json().catch(() => ({}));
   173→        throw new Error(data.error || 'Failed to complete sale');
   174→      }
   175→      const data = await res.json();
   176→      setCompletedSale({
   177→        invoiceNo: data.invoiceNo,
   178→        customerName: data.customer?.name,
   179→        items: (data.items ?? []).map((item: any) => ({
   180→          productName: item.product?.name ?? 'Unknown',
   181→          quantity: item.quantity,
   182→          unitPrice: item.unitPrice,
   183→          total: item.total,
   184→        })),
   185→        subtotal: data.subtotal,
   186→        tax: data.tax,
   187→        discount: data.discount,
   188→        totalAmount: data.totalAmount,
   189→        paymentMethod: data.paymentMethod,
   190→        createdAt: data.createdAt,
   191→      });
   192→      setShowReceipt(true);
   193→      toast.success('Sale completed successfully!');
   194→    } catch (err) {
   195→      toast.error(err instanceof Error ? err.message : 'Failed to complete sale');
   196→    } finally {
   197→      setSubmitting(false);
   198→    }
   199→  };
   200→
   201→  const handleCloseReceipt = () => {
   202→    setShowReceipt(false);
   203→    setCompletedSale(null);
   204→    clearCart();
   205→    setDiscount(0);
   206→    setNotes('');
   207→    setPaymentMethod('cash');
   208→  };
   209→
   210→  const handleAddWalkIn = async () => {
   211→    if (!walkInName.trim()) return;
   212→    setAddingWalkIn(true);
   213→    try {
   214→      const res = await fetch('/api/customers', {
   215→        method: 'POST',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({
   218→          name: walkInName.trim(),
   219→          phone: walkInPhone.trim() || undefined,
   220→        }),
   221→      });
   222→      if (!res.ok) {
   223→        const data = await res.json().catch(() => ({}));
   224→        throw new Error(data.error || 'Failed to add walk-in customer');
   225→      }
   226→      const newCustomer = await res.json();
   227→      setSelectedCustomer(newCustomer.id);
   228→      setCustomerSearch(newCustomer.name);
   229→      setCustomers(prev => [newCustomer, ...prev]);
   230→      setShowWalkInDialog(false);
   231→      toast.success(`${newCustomer.name} added & selected`);
   232→    } catch (err) {
   233→      toast.error(err instanceof Error ? err.message : 'Failed to add walk-in');
   234→    } finally {
   235→      setAddingWalkIn(false);
   236→    }
   237→  };
   238→
   239→  const paymentIcons = {
   240→    cash: Banknote,
   241→    card: CreditCard,
   242→    mobile_money: Smartphone,
   243→  };
   244→
   245→  const selectedCustomer = customers.find(c => c.id === selectedCustomerId);
   246→
   247→  if (loading) {
   248→    return (
   249→      <div className="flex h-full gap-4 p-4">
   250→        <div className="flex-1 space-y-4">
   251→          <Skeleton className="h-10 w-full" />
   252→          <div className="grid grid-cols-2 lg:grid-cols-3 gap-3">
   253→            {Array.from({ length: 9 }).map((_, i) => (
   254→              <Skeleton key={i} className="h-28 w-full" />
   255→            ))}
   256→          </div>
   257→        </div>
   258→        <div className="w-96 border-l p-4 space-y-4">
   259→          <Skeleton className="h-8 w-40" />
   260→          {Array.from({ length: 4 }).map((_, i) => (
   261→            <Skeleton key={i} className="h-16 w-full" />
   262→          ))}
   263→          <Skeleton className="h-40 w-full" />
   264→        </div>
   265→      </div>
   266→    );
   267→  }
   268→
   269→  return (
   270→    <div className="flex h-[calc(100vh-5rem)] gap-4 p-4">
   271→      {/* Left Panel - Product Search & Listing */}
   272→      <div className="flex-1 flex flex-col min-w-0">
   273→        {/* Search */}
   274→        <div className="relative mb-4">
   275→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   276→          <Input
   277→            placeholder="Search products by name..."
   278→            value={searchQuery}
   279→            onChange={(e) => handleSearch(e.target.value)}
   280→            className="pl-10"
   281→          />
   282→        </div>
   283→
   284→        {/* Category Tabs */}
   285→        <div className="flex gap-2 mb-4 overflow-x-auto pb-2">
   286→          {categories.map((cat) => (
   287→            <Button
   288→              key={cat}
   289→              variant={activeCategory === cat ? 'default' : 'outline'}
   290→              size="sm"
   291→              className={activeCategory === cat ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   292→              onClick={() => setActiveCategory(cat)}
   293→            >
   294→              {cat}
   295→            </Button>
   296→          ))}
   297→        </div>
   298→
   299→        {/* Product Grid */}
   300→        <ScrollArea className="flex-1">
   301→          <div className="grid grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   302→            {filteredProducts.length > 0 ? (
   303→              filteredProducts.map((product) => (
   304→                <button
   305→                  key={product.id}
   306→                  onClick={() => handleAddToCart(product)}
   307→                  className="text-left p-4 rounded-xl border hover:border-emerald-400 hover:shadow-md transition-all bg-card"
   308→                >
   309→                  <div className="flex items-start justify-between gap-2">
   310→                    <h4 className="font-medium text-sm leading-tight line-clamp-2">{product.name}</h4>
   311→                    <Pill className="h-4 w-4 text-emerald-500 shrink-0 mt-0.5" />
   312→                  </div>
   313→                  <p className="text-xs text-muted-foreground mt-1">{product.category?.name ?? 'Uncategorized'}</p>
   314→                  <div className="flex items-center justify-between mt-3">
   315→                    <span className="font-bold text-emerald-600">{formatGHS(product.minSellingPrice)}</span>
   316→                    <Badge variant={product.totalStock > 0 ? 'default' : 'destructive'} className={
   317→                      product.totalStock > 0 ? 'bg-emerald-100 text-emerald-700' : ''
   318→                    }>
   319→                      {product.totalStock}
   320→                    </Badge>
   321→                  </div>
   322→                </button>
   323→              ))
   324→            ) : (
   325→              <div className="col-span-full text-center text-muted-foreground py-12">
   326→                No products found
   327→              </div>
   328→            )}
   329→          </div>
   330→        </ScrollArea>
   331→      </div>
   332→
   333→      {/* Right Panel - Cart & Checkout */}
   334→      <div className="w-96 border-l bg-card flex flex-col shrink-0">
   335→        <CardHeader className="pb-3">
   336→          <CardTitle className="text-lg flex items-center gap-2">
   337→            <ShoppingCart className="h-5 w-5" />
   338→            Cart ({cart.length} items)
   339→          </CardTitle>
   340→        </CardHeader>
   341→
   342→        {/* Cart Items */}
   343→        <ScrollArea className="flex-1 px-4">
   344→          {cart.length > 0 ? (
   345→            <div className="space-y-3">
   346→              {cart.map((item) => (
   347→                <div key={`${item.productId}-${item.batchId}`} className="p-3 rounded-lg border">
   348→                  <div className="flex items-start justify-between">
   349→                    <div className="flex-1 min-w-0">
   350→                      <p className="font-medium text-sm truncate">{item.productName}</p>
   351→                      <p className="text-xs text-muted-foreground">{formatGHS(item.unitPrice)} each</p>
   352→                    </div>
   353→                    <Button
   354→                      variant="ghost"
   355→                      size="icon"
   356→                      className="h-7 w-7 text-red-400 hover:text-red-600"
   357→                      onClick={() => removeFromCart(item.productId, item.batchId)}
   358→                    >
   359→                      <Trash2 className="h-4 w-4" />
   360→                    </Button>
   361→                  </div>
   362→                  <div className="flex items-center justify-between mt-2">
   363→                    <div className="flex items-center gap-2">
   364→                      <Button
   365→                        variant="outline"
   366→                        size="icon"
   367→                        className="h-7 w-7"
   368→                        onClick={() => updateCartQuantity(item.productId, item.batchId, item.quantity - 1)}
   369→                        disabled={item.quantity <= 1}
   370→                      >
   371→                        <Minus className="h-3 w-3" />
   372→                      </Button>
   373→                      <span className="w-8 text-center font-medium text-sm">{item.quantity}</span>
   374→                      <Button
   375→                        variant="outline"
   376→                        size="icon"
   377→                        className="h-7 w-7"
   378→                        onClick={() => updateCartQuantity(item.productId, item.batchId, item.quantity + 1)}
   379→                        disabled={item.quantity >= item.availableQty}
   380→                      >
   381→                        <Plus className="h-3 w-3" />
   382→                      </Button>
   383→                    </div>
   384→                    <span className="font-bold text-sm">{formatGHS(item.unitPrice * item.quantity)}</span>
   385→                  </div>
   386→                </div>
   387→              ))}
   388→            </div>
   389→          ) : (
   390→            <div className="text-center text-muted-foreground py-12">
   391→              <ShoppingCart className="h-12 w-12 mx-auto mb-3 opacity-30" />
   392→              <p>Cart is empty</p>
   393→              <p className="text-xs">Click a product to add it</p>
   394→            </div>
   395→          )}
   396→        </ScrollArea>
   397→
   398→        {/* Cart Summary & Checkout */}
   399→        <div className="border-t p-4 space-y-3">
   400→          {/* Customer Selection */}
   401→          <div className="space-y-2">
   402→            <div className="relative">
   403→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   404→              <Input
   405→                placeholder="Search customer (optional)"
   406→                className="pl-8 h-8 text-xs"
   407→                value={customerSearch}
   408→                onChange={(e) => setCustomerSearch(e.target.value)}
   409→              />
   410→              {customerSearch && !selectedCustomerId && (
   411→                <div className="absolute top-full left-0 right-0 mt-1 bg-background border rounded-lg shadow-lg max-h-40 overflow-y-auto z-50">
   412→                  {customers
   413→                    .filter(c => c.name.toLowerCase().includes(customerSearch.toLowerCase()) || c.phone?.includes(customerSearch))
   414→                    .slice(0, 5)
   415→                    .map(c => (
   416→                      <button
   417→                        key={c.id}
   418→                        className="w-full text-left px-3 py-2 hover:bg-muted text-sm"
   419→                        onClick={() => { setSelectedCustomer(c.id); setCustomerSearch(c.name); }}
   420→                      >
   421→                        {c.name} - {c.phone ?? c.email ?? '-'}
   422→                      </button>
   423→                    ))
   424→                }
   425→              </div>
   426→              )}
   427→              {selectedCustomer && (
   428→                <div className="flex items-center justify-between mt-1">
   429→                  <span className="text-xs text-emerald-600 font-medium">{selectedCustomer.name}</span>
   430→                  <button onClick={() => { setSelectedCustomer(null); setCustomerSearch(''); }} className="text-xs text-red-500 hover:underline">Remove</button>
   431→                </div>
   432→              )}
   433→            </div>
   434→            {!selectedCustomerId && (
   435→              <Button
   436→                variant="outline"
   437→                size="sm"
   438→                className="w-full h-8 text-xs border-dashed border-emerald-300 text-emerald-600 hover:bg-emerald-50 hover:text-emerald-700 hover:border-emerald-400 transition-all"
   439→                onClick={() => { setWalkInName('Walk-In Customer'); setWalkInPhone(''); setShowWalkInDialog(true); }}
   440→              >
   441→                <Zap className="h-3 w-3 mr-1" />
   442→                Quick Walk-In
   443→              </Button>
   444→            )}
   445→          </div>
   446→
   447→          {/* Notes */}
   448→          <Input
   449→            placeholder="Notes (optional)"
   450→            value={notes}
   451→            onChange={(e) => setNotes(e.target.value)}
   452→            className="h-8 text-xs"
   453→          />
   454→
   455→          <div className="space-y-1.5 text-sm">
   456→            <div className="flex justify-between">
   457→              <span className="text-muted-foreground">Subtotal</span>
   458→              <span>{formatGHS(subtotal)}</span>
   459→            </div>
   460→            <div className="flex justify-between">
   461→              <span className="text-muted-foreground">Tax (12.5%)</span>
   462→              <span>{formatGHS(tax)}</span>
   463→            </div>
   464→            <div className="flex items-center justify-between">
   465→              <span className="text-muted-foreground">Discount</span>
   466→              <Input
   467→                type="number"
   468→                value={discount || ''}
   469→                onChange={(e) => setDiscount(Number(e.target.value) || 0)}
   470→                className="w-24 h-7 text-xs text-right"
   471→                placeholder="0.00"
   472→              />
   473→            </div>
   474→            <Separator />
   475→            <div className="flex justify-between font-bold text-lg">
   476→              <span>Total</span>
   477→              <span className="text-emerald-600">{formatGHS(total)}</span>
   478→            </div>
   479→          </div>
   480→
   481→          {/* Payment Method */}
   482→          <div className="flex gap-2">
   483→            {(['cash', 'card', 'mobile_money'] as const).map((method) => {
   484→              const Icon = paymentIcons[method];
   485→              const label = method === 'mobile_money' ? 'Mobile Money' : method.charAt(0).toUpperCase() + method.slice(1);
   486→              return (
   487→                <Button
   488→                  key={method}
   489→                  variant={paymentMethod === method ? 'default' : 'outline'}
   490→                  size="sm"
   491→                  className={paymentMethod === method ? 'bg-emerald-600 hover:bg-emerald-700 flex-1' : 'flex-1'}
   492→                  onClick={() => setPaymentMethod(method)}
   493→                >
   494→                  <Icon className="h-4 w-4 mr-1" />
   495→                  {label}
   496→                </Button>
   497→              );
   498→            })}
   499→          </div>
   500→
   501→          {/* Action Buttons */}
   502→          <div className="flex gap-2">
   503→            <Button
   504→              variant="outline"
   505→              className="flex-1"
   506→              onClick={clearCart}
   507→              disabled={cart.length === 0}
   508→            >
   509→              <X className="h-4 w-4 mr-1" />
   510→              Clear
   511→            </Button>
   512→            <Button
   513→              className="flex-1 bg-emerald-600 hover:bg-emerald-700 text-white"
   514→              onClick={handleCompleteSale}
   515→              disabled={cart.length === 0 || submitting}
   516→            >
   517→              <CheckCircle className="h-4 w-4 mr-1" />
   518→              {submitting ? 'Processing...' : 'Complete Sale'}
   519→            </Button>
   520→          </div>
   521→        </div>
   522→      </div>
   523→
   524→      {/* Walk-In Customer Dialog */}
   525→      <Dialog open={showWalkInDialog} onOpenChange={(open) => { if (!open) setShowWalkInDialog(false); }}>
   526→        <DialogContent className="sm:max-w-sm">
   527→          <DialogHeader>
   528→            <DialogTitle className="flex items-center gap-2">
   529→              <div className="h-8 w-8 rounded-lg bg-emerald-100 flex items-center justify-center">
   530→                <UserRoundPlus className="h-4 w-4 text-emerald-600" />
   531→              </div>
   532→              Quick Walk-In
   533→            </DialogTitle>
   534→          </DialogHeader>
   535→          <div className="space-y-3">
   536→            <div>
   537→              <Label className="text-xs text-muted-foreground">Customer Name</Label>
   538→              <Input
   539→                value={walkInName}
   540→                onChange={(e) => setWalkInName(e.target.value)}
   541→                placeholder="e.g. Walk-In Customer"
   542→                className="mt-1"
   543→                autoFocus
   544→              />
   545→            </div>
   546→            <div>
   547→              <Label className="text-xs text-muted-foreground">Phone <span className="text-muted-foreground/50">(optional)</span></Label>
   548→              <Input
   549→                value={walkInPhone}
   550→                onChange={(e) => setWalkInPhone(e.target.value)}
   551→                placeholder="e.g. +233 24 123 4567"
   552→                className="mt-1"
   553→                onKeyDown={(e) => { if (e.key === 'Enter') handleAddWalkIn(); }}
   554→              />
   555→            </div>
