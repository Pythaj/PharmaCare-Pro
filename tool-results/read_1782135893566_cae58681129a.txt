    82→// CountingNumber – animated digital counter / slot-machine number
    83→// ─────────────────────────────────────────────────────────────
    84→function CountingNumber({
    85→  value,
    86→  direction,
    87→  className = '',
    88→  size = 'sm',
    89→}: {
    90→  value: number;
    91→  direction?: 'up' | 'down' | null;
    92→  className?: string;
    93→  size?: 'sm' | 'md' | 'lg' | 'xl';
    94→}) {
    95→  const motionVal = useMotionValue(value);
    96→  const display = useTransform(motionVal, (v) => Math.round(v));
    97→  const springVal = useSpring(display, { stiffness: 300, damping: 30 });
    98→  const rounded = useTransform(springVal, (v) => Math.round(v));
    99→
   100→  useEffect(() => {
   101→    const controls = animate(motionVal, value, {
   102→      duration: 0.6,
   103→      ease: 'easeOut',
   104→    });
   105→    return () => controls.stop();
   106→  }, [value, motionVal]);
   107→
   108→  const sizeClasses = {
   109→    sm: 'text-sm',
   110→    md: 'text-lg',
   111→    lg: 'text-2xl',
   112→    xl: 'text-4xl',
   113→  };
   114→
   115→  const colorClass = direction === 'up'
   116→    ? 'text-emerald-500'
   117→    : direction === 'down'
   118→    ? 'text-red-500'
   119→    : 'text-inherit';
   120→
   121→  return (
   122→    <motion.span
   123→      className={`font-bold tabular-nums font-mono tracking-tighter ${sizeClasses[size]} ${colorClass} ${className}`}
   124→      key={`${value}-${direction}`}
   125→      initial={direction ? { scale: 1.35 } : false}
   126→      animate={{ scale: 1 }}
   127→      transition={{ type: 'spring', stiffness: 400, damping: 20 }}
   128→    >
   129→      <motion.span>{rounded}</motion.span>
   130→    </motion.span>
   131→  );
   132→}

   134→// ─────────────────────────────────────────────────────────────
   135→// StockGauge – circular SVG ring gauge
   136→// ─────────────────────────────────────────────────────────────
   137→function StockGauge({
   138→  stock,
   139→  reorderLevel,
   140→  size = 40,
   141→  strokeWidth = 3,
   142→}: {
   143→  stock: number;
   144→  reorderLevel: number;
   145→  size?: number;
   146→  strokeWidth?: number;
   147→}) {
   148→  const radius = (size - strokeWidth) / 2;
   149→  const circumference = 2 * Math.PI * radius;
   150→  const maxDisplay = Math.max(reorderLevel * 3, 20);
   151→  const pct = stock === 0 ? 0 : Math.min((stock / maxDisplay) * 100, 100);
   152→  const isCritical = stock > 0 && stock <= Math.ceil(reorderLevel * 0.3);
   153→  const isLow = stock > 0 && stock <= reorderLevel;
   154→  const isOut = stock === 0;
   155→
   156→  const color = isOut ? '#ef4444' : isCritical ? '#ef4444' : isLow ? '#f59e0b' : '#10b981';
   157→  const bgColor = isOut ? '#fecaca' : isCritical ? '#fecaca' : isLow ? '#fef3c7' : '#d1fae5';
   158→
   159→  const animatedOffset = useMotionValue(circumference);
   160→  useEffect(() => {
   161→    animate(animatedOffset, circumference - (pct / 100) * circumference, {
   162→      duration: 0.8,
   163→      ease: 'easeOut',
   164→    });
   165→  }, [pct, circumference, animatedOffset]);
   166→
   167→  const strokeDashoffset = useTransform(animatedOffset, (v) => v);
   168→
   169→  if (isOut) {
   170→    return (
   171→      <div className="relative" style={{ width: size, height: size }}>
   172→        <svg width={size} height={size} className="-rotate-90">
   173→          <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={bgColor} strokeWidth={strokeWidth} />
   174→          <motion.circle
   175→            cx={size / 2}
   176→            cy={size / 2}
   177→            r={radius}
   178→            fill="none"
   179→            stroke={color}
   180→            strokeWidth={strokeWidth}
   181→            strokeLinecap="round"
   182→            strokeDasharray={circumference}
   183→            style={{ strokeDashoffset }}
   184→          />
   185→        </svg>
   186→        <div className="absolute inset-0 flex items-center justify-center">
   187→          <X className="text-red-500" style={{ width: size * 0.32, height: size * 0.32 }} strokeWidth={3} />
   188→        </div>
   189→      </div>
   190→    );
   191→  }
   192→
   193→  return (
   194→    <div className="relative" style={{ width: size, height: size }}>
   195→      <svg width={size} height={size} className="-rotate-90">
   196→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={bgColor} strokeWidth={strokeWidth} />
   197→        <motion.circle
   198→          cx={size / 2}
   199→          cy={size / 2}
   200→          r={radius}
   201→          fill="none"
   202→          stroke={color}
   203→          strokeWidth={strokeWidth}
   204→          strokeLinecap="round"
   205→          strokeDasharray={circumference}
   206→          style={{ strokeDashoffset }}
   207→        />
   208→        {isCritical && (
   209→          <motion.circle
   210→            cx={size / 2}
   211→            cy={size / 2}
   212→            r={radius}
   213→            fill="none"
   214→            stroke={color}
   215→            strokeWidth={strokeWidth + 1}
   216→            strokeLinecap="round"
   217→            strokeDasharray={circumference}
   218→            initial={{ opacity: 0.8 }}
   219→            animate={{ opacity: [0.8, 0, 0.8] }}
   220→            transition={{ repeat: Infinity, duration: 1.5, ease: 'easeInOut' }}
   221→            style={{ strokeDashoffset: circumference - (pct / 100) * circumference }}
   222→          />
   223→        )}
   224→      </svg>
   225→      <div className="absolute inset-0 flex items-center justify-center">
   226→        <span
   227→          className="font-bold tabular-nums font-mono"
   228→          style={{
   229→            fontSize: size * 0.3,
   230→            color: color,
   231→          }}
   232→        >
   233→          {stock}
   234→        </span>
   235→      </div>
   236→    </div>
   237→  );
   238→}

   306→function StockImpactModal({
   307→  items,
   308→  onContinue,
   309→}: {
   310→  items: StockImpactItem[];
   311→  onContinue: () => void;
   312→}) {
   313→  const [visibleItems, setVisibleItems] = useState<number>(0);

   316→    if (visibleItems < items.length) {
   317→      const timer = setTimeout(() => setVisibleItems((v) => v + 1), 200);
   318→      return () => clearTimeout(timer);
   319→    }
   320→  }, [visibleItems, items.length]);
   321→
   322→  return (
   323→    <Dialog open onOpenChange={() => {}}>
   324→      <DialogContent className="max-w-lg p-0 gap-0 overflow-hidden" showCloseButton={false}>
   325→        <DialogTitle className="sr-only">Stock Impact Report</DialogTitle>
   326→        <DialogDescription className="sr-only">Inventory has been updated for this sale</DialogDescription>
   327→        {/* Header */}
   328→        <div className="bg-gradient-to-r from-slate-900 to-slate-800 px-6 py-4 text-white">
   329→          <div className="flex items-center gap-3">
   330→            <motion.div
   331→              className="h-10 w-10 rounded-xl bg-emerald-500/20 flex items-center justify-center"
   332→              animate={{ rotate: [0, 10, -10, 0] }}
   333→              transition={{ duration: 0.5, delay: 0.2 }}
   334→            >
   335→              <Package className="h-5 w-5 text-emerald-400" />
   336→            </motion.div>
   337→            <div>
   338→              <h2 className="font-bold text-lg">Stock Impact Report</h2>
   339→              <p className="text-xs text-slate-400">Inventory has been updated for this sale</p>
   340→            </div>
   341→          </div>
   342→        </div>
   343→
   344→        {/* Items */}
   345→        <div className="px-4 py-4 max-h-[60vh] overflow-y-auto space-y-3">
   346→          {items.map((item, idx) => {
   347→            const maxBar = Math.max(item.oldStock, item.reorderLevel * 3, 10);
   348→            const oldPct = (item.oldStock / maxBar) * 100;
   349→            const newPct = (item.newStock / maxBar) * 100;
   350→            const barColor = item.hitZero ? '#ef4444' : item.hitLow ? '#f59e0b' : '#10b981';
   351→
   352→            return (
   353→              <motion.div
   354→                key={item.productId}
   355→                initial={{ opacity: 0, y: 20 }}
   356→                animate={visibleItems > idx ? { opacity: 1, y: 0 } : {}}
   357→                transition={{ type: 'spring', stiffness: 300, damping: 25 }}
   358→                className={`relative rounded-xl border-2 p-4 overflow-hidden ${
   359→                  item.hitZero
   360→                    ? 'border-red-200 bg-red-50/50'
   361→                    : item.hitLow
   362→                    ? 'border-amber-200 bg-amber-50/30'
   363→                    : 'border-slate-100 bg-white'
   364→                }`}
   365→              >
   366→                {/* OUT OF STOCK stamp */}
   367→                <AnimatePresence>
   368→                  {item.hitZero && visibleItems > idx && (
   369→                    <motion.div
   370→                      initial={{ rotate: -15, scale: 0, opacity: 0 }}
   371→                      animate={{ rotate: -15, scale: 1, opacity: 1 }}
   372→                      transition={{ delay: 0.4, type: 'spring', stiffness: 400, damping: 15 }}
   373→                      className="absolute top-3 right-3 pointer-events-none"

   375→                      <div className="border-3 border-red-400 text-red-500 px-3 py-1 rounded-md font-black text-xs uppercase tracking-widest bg-red-50/80"

   377→                        OUT OF STOCK
   378→                      </div>
   379→                    </motion.div>
   380→                  )}
   381→                </AnimatePresence>
   382→
   383→                {/* LOW STOCK alert */}
   384→                <AnimatePresence>
   385→                  {item.hitLow && !item.hitZero && visibleItems > idx && (
   386→                    <motion.div
   387→                      initial={{ opacity: 0, x: 20 }}
   388→                      animate={{ opacity: 1, x: 0 }}
   389→                      transition={{ delay: 0.3 }}
   390→                      className="absolute top-3 right-3"
   391→                    >
   392→                      <motion.div
   393→                        animate={{ opacity: [1, 0.5, 1] }}
   394→                        transition={{ repeat: Infinity, duration: 1.2 }}
   395→                        className="flex items-center gap-1 text-amber-600 text-[10px] font-bold bg-amber-100 px-2 py-0.5 rounded-full"

   397→                        <AlertTriangle className="h-3 w-3" />
   398→                        LOW STOCK ALERT
   399→                      </motion.div>
   400→                    </motion.div>
   401→                  )}

   403→

   405→                  <p className="font-semibold text-sm text-slate-800 mb-1">{item.productName}</p>
   406→                  <p className="text-xs text-slate-400 mb-3">Sold: {item.quantitySold} unit{item.quantitySold > 1 ? 's' : ''}</p>
   407→
   408→                  {/* Animated counter row */}
   409→                  <div className="flex items-center gap-3 mb-3">
   410→                    <div className="text-center">
   411→                      <p className="text-[10px] text-slate-400 mb-0.5">Before</p>
