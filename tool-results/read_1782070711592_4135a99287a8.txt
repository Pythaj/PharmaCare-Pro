    38→function formatGHS(value: number): string {
    39→  return new Intl.NumberFormat('en-GH', { style: 'currency', currency: 'GHS' }).format(value);
    40→}

    42→function getInitials(name: string): string {
    43→  return name
    44→    .split(' ')
    45→    .map((w) => w[0])
    46→    .join('')
    47→    .toUpperCase()
    48→    .slice(0, 2);
    49→}

    51→function todayStr(): string {
    52→  const d = new Date();
    53→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    54→}

    56→function monthStartStr(): string {
    57→  const d = new Date();
